rp_idiom.0001 = {
	type = character_event
	title = rp_idiom.0001.t
	theme = martial
	override_background = { reference = tgp_camp_asia }
	override_icon = { reference = "gfx/interface/icons/event_types/type_war.dds" }
	override_icon = { reference = "gfx/interface/icons/event_types/type_war.dds" }
	desc = rp_idiom.0001.desc
	right_portrait = scope:rp_idiom_enemy

	trigger = {
		is_at_war = yes
		exists = scope:rp_idiom_enemy
		OR = {
			has_relation_friend = scope:rp_idiom_enemy
			has_relation_best_friend = scope:rp_idiom_enemy
		}
	}

	option = {
		name = rp_idiom.0001.a
		add_prestige = 100
		add_character_modifier = { modifier = rp_idiom_retreat_righteous_modifier years = 3 }
		if = {
			limit = { exists = scope:rp_idiom_enemy }
			scope:rp_idiom_enemy = {
				add_opinion = { target = root modifier = rp_idiom_ally_grateful_opinion }
			}
		}
	}

	option = {
		name = rp_idiom.0001.b
		add_prestige = -25
		add_character_modifier = { modifier = rp_idiom_battle_resolve_modifier years = 2 }
	}

	option = {
		name = rp_idiom.0001.c
		random_list = {
			55 = {
				add_prestige = 75
				add_character_modifier = { modifier = rp_idiom_ambush_success_modifier years = 2 }
			}
			45 = {
				add_prestige = -50
				add_character_modifier = { modifier = rp_idiom_ambush_failure_modifier years = 2 }
			}
		}
	}

	option = {
		name = rp_idiom.0001.d
		trigger = {
			scope:rp_idiom_enemy = {
				OR = {
					has_relation_friend = root
					has_relation_best_friend = root
				}
			}
			rp_has_gentle_disposition_trigger = yes
		}
		add_prestige = 50
		add_character_modifier = { modifier = rp_idiom_retreat_righteous_modifier years = 5 }
		scope:rp_idiom_enemy = {
			add_opinion = { target = root modifier = rp_idiom_ally_grateful_opinion }
		}
	}

	option = {
		name = rp_idiom.0001.e
		trigger = {
			rp_has_forceful_disposition_trigger = yes
		}
		add_prestige = 25
		add_dread = 5
		add_character_modifier = { modifier = rp_idiom_battle_resolve_modifier years = 3 }
	}
}
