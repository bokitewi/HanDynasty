namespace = rp_hanxin

rp_hanxin.0005 = {
	type = character_event
	title = rp_hanxin.0005.t
	theme = intrigue
	override_background = { reference = tgp_study_asia }
	override_icon = { reference = "gfx/interface/icons/event_types/type_intrigue.dds" }
	override_icon = { reference = "gfx/interface/icons/event_types/type_intrigue.dds" }
	desc = rp_hanxin.0005.desc
	right_portrait = scope:rp_hanxin_actor

	trigger = {
		exists = scope:rp_hanxin_actor
		scope:rp_hanxin_actor = {
			NOT = { has_character_flag = rp_hanxin_recent_rumor }
			has_variable = hanxin_ambition
			var:hanxin_ambition >= 40
		}
	}

	immediate = {
		scope:rp_hanxin_actor = {
			add_character_flag = { flag = rp_hanxin_recent_rumor days = 1095 }
		}
	}

	option = {
		name = rp_hanxin.0005.a
		scope:rp_hanxin_actor = {
			change_variable = { name = hanxin_ambition add = -10 }
			if = {
				limit = { var:hanxin_ambition < 20 }
				set_variable = { name = hanxin_ambition value = 20 }
			}
			add_character_modifier = { modifier = rp_power_under_suspicion_modifier years = 3 }
			add_opinion = { target = root modifier = rp_merit_legal_rebuff_opinion }
		}
	}

	option = {
		name = rp_hanxin.0005.b
		trigger = {
			scope:rp_hanxin_actor = {
				OR = {
					rp_hanxin_child_ward_with_liege_trigger = { LIEGE = root }
					rp_hanxin_close_family_at_liege_court_trigger = { LIEGE = root }
				}
			}
		}
		scope:rp_hanxin_actor = {
			change_variable = { name = hanxin_ambition add = -15 }
			if = {
				limit = { var:hanxin_ambition < 20 }
				set_variable = { name = hanxin_ambition value = 20 }
			}
			add_opinion = { target = root modifier = rp_merit_persuaded_opinion }
		}
	}

	option = {
		name = rp_hanxin.0005.c
		scope:rp_hanxin_actor = {
			rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = 5 }
		}
	}

	option = {
		name = rp_hanxin.0005.d
		trigger = {
			gold >= 25
			OR = {
				rp_has_gentle_disposition_trigger = yes
				scope:rp_hanxin_actor = {
					rp_has_gentle_disposition_trigger = yes
				}
			}
		}
		remove_short_term_gold = 25
		scope:rp_hanxin_actor = {
			add_opinion = { target = root modifier = rp_merit_reward_honor_opinion }
			add_character_modifier = { modifier = rp_merit_honored_modifier years = 3 }
			rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = -15 }
		}
	}

	option = {
		name = rp_hanxin.0005.e
		trigger = {
			OR = {
				rp_has_cunning_disposition_trigger = yes
				scope:rp_hanxin_actor = {
					rp_has_cunning_disposition_trigger = yes
				}
			}
		}
		scope:rp_hanxin_actor = {
			add_character_modifier = { modifier = rp_power_under_suspicion_modifier years = 3 }
			random_list = {
				70 = { rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = -5 } }
				30 = { rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = 10 } }
			}
		}
		add_prestige = 15
	}

	option = {
		name = rp_hanxin.0005.f
		trigger = {
			rp_has_forceful_disposition_trigger = yes
		}
		add_dread = 10
		scope:rp_hanxin_actor = {
			add_opinion = { target = root modifier = rp_merit_refused_harsh_opinion }
			random_list = {
				60 = { rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = -20 } }
				40 = { rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = 15 } }
			}
		}
	}
}

rp_hanxin.0100 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = var:rp_hanxin_ambition_target }
			var:rp_hanxin_ambition_target = { save_scope_as = rp_hanxin_actor }
		}
		if = {
			limit = {
				exists = scope:rp_hanxin_actor
				scope:rp_hanxin_actor = {
					rp_hanxin_ambition_actor_valid_trigger = { LIEGE = root }
				}
			}
			random_list = {
				35 = { trigger_event = rp_hanxin.0110 }
				45 = { trigger_event = rp_hanxin.0120 }
				20 = { trigger_event = rp_hanxin.0130 }
			}
		}
		else = {
			rp_clear_hanxin_ambition_chain_effect = yes
		}
	}
}
