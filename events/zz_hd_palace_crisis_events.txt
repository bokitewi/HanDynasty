namespace = hd_palace_crisis

# A single viewpoint is selected by the Yellow Turban result effect.  The
# viewpoint can be He Jin, Yuan Shao, or Cao Cao, but the historical sequence
# remains the same for every player.
hd_palace_crisis.0001 = {
	type = character_event
	title = hd_palace_crisis.0001.t
	desc = hd_palace_crisis.0001.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_cynical
	}

	trigger = {
		exists = global_var:hd_palace_crisis_started
		is_alive = yes
		OR = {
			this = character:he_jin
			this = character:yuan_shao
			this = character:cao_cao
		}
	}

	immediate = {
		add_character_flag = hd_palace_crisis_controller
	}

	option = {
		name = hd_palace_crisis.0001.a
		trigger_event = {
			id = hd_palace_crisis.0002
			days = { 7 14 }
		}
		ai_chance = { base = 100 }
	}
}

hd_palace_crisis.0002 = {
	type = character_event
	title = hd_palace_crisis.0002.t
	desc = hd_palace_crisis.0002.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_rational
	}

	trigger = {
		exists = global_var:hd_palace_crisis_started
		is_alive = yes
		has_character_flag = hd_palace_crisis_controller
	}

	option = {
		name = hd_palace_crisis.0002.a
		if = {
			limit = { title:k_uuii_liangzhou.holder ?= { is_alive = yes } }
			title:k_uuii_liangzhou.holder = {
				trigger_event = {
					id = hd_palace_crisis.0003
					days = { 7 14 }
				}
			}
		}
		else = {
			# If Liangzhou has already lost its holder, the palace still follows
			# the historical He Jin murder but simply has no outside army to call.
			character:he_jin ?= {
				trigger_event = {
					id = hd_palace_crisis.0004
					days = { 7 14 }
				}
			}
		}
		ai_chance = { base = 100 }
	}
}

				id = hd_palace_crisis.0004

			}

}

hd_palace_crisis.0004 = {
	type = character_event
	title = hd_palace_crisis.0004.t
	desc = hd_palace_crisis.0004.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = personality_coward
	}

	trigger = {
		exists = global_var:hd_palace_crisis_started
		this = character:he_jin
		is_alive = yes
		NOT = { exists = global_var:hd_palace_he_jin_killed }
	}

	immediate = {
		character:zhang_rang ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = hd_palace_killer
			}
		}
		if = {
			limit = { NOT = { exists = scope:hd_palace_killer } }
			character:zhao_zhong ?= {
				if = {
					limit = { is_alive = yes }
					save_scope_as = hd_palace_killer
				}
			}
		}
	}

	option = {
		name = hd_palace_crisis.0004.a
		set_global_variable = hd_palace_he_jin_killed

		# Yuan Shao is the first commander to hear the news if he is a player;
		# Cao Cao is the second choice.  AI commanders use the same order.
		if = {
			limit = { character:yuan_shao ?= { is_alive = yes is_local_player = yes } }
			character:yuan_shao = { trigger_event = { id = hd_palace_crisis.0005 days = { 7 14 } } }
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes is_local_player = yes } }
			character:cao_cao = { trigger_event = { id = hd_palace_crisis.0005 days = { 7 14 } } }
		}
		else_if = {
			limit = { character:yuan_shao ?= { is_alive = yes } }
			character:yuan_shao = { trigger_event = { id = hd_palace_crisis.0005 days = { 7 14 } } }
		}
		else_if = {
			limit = { character:cao_cao ?= { is_alive = yes } }
			character:cao_cao = { trigger_event = { id = hd_palace_crisis.0005 days = { 7 14 } } }
		}

		if = {
			limit = { exists = scope:hd_palace_killer }
			death = {
				death_reason = death_murder
				killer = scope:hd_palace_killer
			}
		}
		else = {
			death = { death_reason = death_murder }
		}
		ai_chance = { base = 100 }
	}
}

hd_palace_crisis.0005 = {
	type = character_event
	title = hd_palace_crisis.0005.t
	desc = hd_palace_crisis.0005.desc
	theme = war

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		exists = global_var:hd_palace_he_jin_killed
		is_alive = yes
		OR = {
			this = character:yuan_shao
			this = character:cao_cao
		}
	}

	immediate = {
		save_scope_as = hd_palace_malefactor
	}

	option = {
		name = hd_palace_crisis.0005.a

		character:zhang_rang ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:zhao_zhong ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:duan_gui ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:guo_sheng ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:xia_yun ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:gao_wang ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:sun_zhang ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:li_song_4 ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:han_kui ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}
		character:zhang_gong ?= {
			if = { limit = { is_alive = yes } death = { death_reason = death_execution killer = scope:hd_palace_malefactor } }
		}

		if = {
			limit = { title:k_uuii_liangzhou = { has_variable = hd_palace_liangzhou_invited } }
			if = {
				limit = { title:k_uuii_liangzhou.holder ?= { is_alive = yes } }
				title:k_uuii_liangzhou.holder = {
					trigger_event = {
						id = hd_palace_crisis.0006
						months = 1
					}
				}
			}
			else = {
				title:k_uuii_liangzhou = { remove_variable = hd_palace_liangzhou_invited }
				set_global_variable = hd_palace_crisis_completed
			}
		}
		else = {
			set_global_variable = hd_palace_crisis_completed
		}
		ai_chance = { base = 100 }
	}
}

hd_palace_crisis.0006 = {
	type = character_event
	title = hd_palace_crisis.0006.t
	desc = hd_palace_crisis.0006.desc
	theme = war

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		exists = global_var:hd_palace_he_jin_killed
		this = title:k_uuii_liangzhou.holder
		is_alive = yes
		title:k_uuii_liangzhou = { has_variable = hd_palace_liangzhou_invited }
	}

	option = {
		name = hd_palace_crisis.0006.a
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = hd_palace_sili_change
			add_claim_on_loss = no
		}
		title:k_uuii_sili = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:hd_palace_sili_change
			}
		}
		resolve_title_and_vassal_change = scope:hd_palace_sili_change
		title:k_uuii_liangzhou = { remove_variable = hd_palace_liangzhou_invited }
		set_global_variable = hd_palace_crisis_completed
		ai_chance = { base = 100 }
	}
}
