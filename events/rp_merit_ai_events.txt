namespace = rp_merit_ai

rp_merit_ai.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		add_character_flag = {
			flag = rp_war_demand_event_cooldown
			days = 1095
		}
		rp_select_war_demand_actor_effect = yes
		if = {
			limit = { exists = scope:rp_war_demand_actor }
			scope:rp_war_demand_actor = {
				rp_add_merit_effect = { LIEGE = root VALUE = 6 }
				rp_hanxin_event_ambition_effect = { LIEGE = root VALUE = 10 }
				add_character_modifier = {
					modifier = rp_campaign_discipline_modifier
					years = 3
				}
				add_opinion = {
					target = root
					modifier = rp_campaign_incident_grateful_opinion
				}
			}
			add_prestige = 30
		}
	}
}

rp_merit_ai.0002 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = var:rp_hanxin_ambition_target }

				rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = 10 }
			}
		}
		else = {
			rp_select_hanxin_ambition_actor_effect = yes
			if = {
				limit = { exists = scope:rp_hanxin_actor }
				rp_activate_selected_hanxin_ambition_actor_effect = yes
				scope:rp_hanxin_actor = {
					rp_hanxin_add_ambition_effect = { LIEGE = root VALUE = 10 }
				}
			}
		}
	}
}

rp_merit_ai.0007 = {
	type = character_event
	hidden = yes

	immediate = {
		rp_select_merit_atonement_actor_effect = yes
		if = {
			limit = { exists = scope:rp_atonement_actor }
			scope:rp_atonement_actor = {
				if = {
					limit = { gold >= 80 }
					random = {
						chance = 45
						remove_short_term_gold = 80
						root = { add_gold = 60 }
						rp_complete_merit_atonement_effect = { LIEGE = root VALUE = 3 }
					}
				}
				else_if = {
					limit = {
						OR = {
							martial >= 12
							prowess >= 14
							has_trait = brave
							has_trait = diligent
						}
					}
					random = {
						chance = 60
						add_character_flag = {
							flag = rp_atone_by_war_service
							days = 1825
						}
						add_character_flag = {
							flag = rp_merit_atonement_recently_handled
							days = 1095
						}
						add_character_modifier = {
							modifier = rp_atonement_border_service_modifier
							years = 5
						}
						add_opinion = {
							target = root
							modifier = rp_merit_atonement_accepted_opinion
						}
					}
				}
				else = {
					random_list = {
						60 = {
							add_character_modifier = {
								modifier = rp_atonement_border_service_modifier
								years = 3
							}
							rp_complete_merit_atonement_effect = { LIEGE = root VALUE = 4 }
						}
						40 = {
							rp_fail_merit_atonement_effect = { LIEGE = root }
						}
					}
				}
			}
		}
	}
}
