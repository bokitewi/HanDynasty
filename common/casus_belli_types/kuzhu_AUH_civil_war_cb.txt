kuzhu_install_warlord_faction_war = {
	icon = install_warlord_faction_war
	group = civil_war
	ai_only_against_liege = yes
	target_titles = independence_domain
	allow_hostages = no
	allowed_for_character = {
		scope:attacker = {
			is_leading_faction_type = kuzhu_install_warlord_faction
			NOT = {
				government_has_flag = government_is_landless_adventurer
			}
		}
	}

	allowed_against_character = {
		scope:attacker = {
			liege = scope:defender
		}
	}

	target_de_jure_regions_above = yes

	valid_to_start = {

	}

	on_declaration = {
		#on_declared_war = yes
		# Remove offending HumSac modifiers.
		scope:attacker = {
			hidden_effect = { fp1_remove_humsac_offended_counties_effect = yes }
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { # Desc for only one player attacker
					scope:attacker = {
						is_local_player = yes
						joined_faction = {
							any_faction_member = {
								count = 1
							}
						}
					}
				}
				desc = kuzhu_install_warlord_war_victory_desc_local_player_attacker_alone
			}
			triggered_desc = { # Desc for only one attacker, player defender
				trigger = {
					scope:defender = {
						is_local_player = yes
					}
					scope:attacker = {
						joined_faction = {
							any_faction_member = {
								count = 1
							}
						}
					}
				}
				desc = kuzhu_install_warlord_war_victory_desc_local_player_defender_attacker_alone
			}
			triggered_desc = { # Desc for player attacker
				trigger = {
					scope:attacker.joined_faction = {
						any_faction_member = { is_local_player = yes }
					}
				}
				desc = kuzhu_install_warlord_war_victory_desc_local_player_attacker
			}
			triggered_desc = { # Desc for player defender
				trigger = { scope:defender = { is_local_player = yes } }
				desc = kuzhu_install_warlord_war_victory_desc_local_player_defender
			}
			desc = kuzhu_install_warlord_war_victory_desc # Desc for a third party involved
		}

	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain from winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		scope:attacker.joined_faction = {
			every_faction_member = {
				#苦竹：简单转为军阀
				kuzhu_convert_to_warlord_from_admin_effect = yes
				add_character_flag = {
					flag = flag_kuzhu_new_warlord
					years = 10
				}
				hidden_effect = {
					add_truce_both_ways = {
						character = scope:defender
						days = 1825
						war = root.war
						result = victory
					}
					save_scope_as = current_member
				}
			}
		}

		scope:attacker = {
			add_prestige = medium_prestige_value
		}
		scope:defender = {
			add_prestige = {
				value = medium_prestige_value
				multiply = -1
			}
		}

		# LEGITIMACY FROM LOSING FACTION WAR
		faction_war_end_defeat_legitimacy_effect = yes

		# EP3: note gold gained from military assistance/join war contracts and their war contribution threshold
		laamp_as_mercenary_payout_tooltip_effect = yes

		#Mandalas gain or lose piety/devotion depending on Decree
		mandala_war_victory_effects = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = kuzhu_install_warlord_war_white_peace_defender_desc
			}
			triggered_desc = {
				trigger = {
					scope:attacker.joined_faction = {
						any_faction_member = { is_local_player = yes }
					}
				}
				desc = kuzhu_install_warlord_war_white_peace_attacker_desc
			}
			desc = kuzhu_install_warlord_war_white_peace_desc
		}
		desc = kuzhu_install_warlord_war_white_peace_end_desc

	}

	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			stress_impact = {
 				ambitious = medium_stress_impact_gain
 				arrogant = medium_stress_impact_gain
 			}
		}
		hidden_effect = {
			scope:attacker = {
				add_truce_both_ways = {
					character = scope:defender
					days = 1825
					war = root.war
					result = white_peace
				}
			}
		}

		scope:defender = {
			stress_impact = {
 				arrogant = medium_stress_impact_gain
 			}

			add_character_flag = {
				flag = kuzhu_recent_install_warlord_faction_war
				years = faction_war_white_peace_cooldown
			}
			add_prestige = minor_prestige_value
		}

		on_white_peace_faction_revolt_war = yes

		# EP3: note gold gained from military assistance/join war contracts and their war contribution threshold
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = kuzhu_install_warlord_war_defeat_defender_desc
			}
			triggered_desc = {
				trigger = {
					scope:attacker.joined_faction = {
						any_faction_member = { is_local_player = yes }
					}
				}
				desc = kuzhu_install_warlord_war_defeat_attacker_desc
			}
			desc = kuzhu_install_warlord_war_defeat_desc #same loc as for player
		}
		desc = kuzhu_install_warlord_war_defeat_end_desc
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			add_character_flag = {
				flag = kuzhu_recent_install_warlord_faction_war
				years = faction_war_defeat_cooldown
			}
			add_dread = medium_dread_gain
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
			add_achievement_flag_effect = { FLAG = achievement_know_your_place_flag }

			# LEGITIMACY FROM WINNING FACTION WAR
			faction_war_end_victory_legitimacy_effect = yes
		}
		on_lost_faction_revolt_war = yes

		# EP3: note gold gained from military assistance/join war contracts and their war contribution threshold
		laamp_as_mercenary_payout_tooltip_effect = yes

		#Mandalas gain or lose piety/devotion depending on Decree
		mandala_war_defeat_effects = yes
	}

	on_invalidated_desc = msg_invalidate_war_title

	check_defender_inheritance_validity = no

	on_primary_attacker_death = inherit_faction
	on_primary_defender_death = inherit

	transfer_behavior = transfer

	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "KUZHU_INSTALL_WARLORD_FACTION_WAR_NAME"

	interface_priority = 80

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 1.0
	defender_wargoal_percentage = 0.0 # A single occupation will do
	defender_ticking_warscore_delay = { days = 0 } # No need for a delay here since the defender actually needs to occupy something rather than starting in control

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50
	
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150

	max_ai_diplo_distance_to_title = 500
}

kuzhu_install_puppet_faction_war = {
	icon = install_puppet_faction_war
	group = civil_war
	allow_hostages = no

	target_top_liege_if_outside_realm = no # In case of Adventurers starting this war

	allowed_for_character = {
		scope:attacker = {
			is_leading_faction_type = kuzhu_install_puppet_faction
		}
	}

	allowed_against_character = {
		scope:attacker = {
			top_liege = scope:defender
		}
	}

	on_declaration = {
		on_declared_war = yes
		# Remove offending HumSac modifiers.
		scope:attacker = {
			hidden_effect = { fp1_remove_humsac_offended_counties_effect = yes }
		}
	}

	on_victory_desc = {
		first_valid = {
			# Desc for player attacker
			triggered_desc = {
				trigger = { 
					scope:attacker = {
						is_local_player = yes
					}
				}
				desc = kuzhu_install_puppet_war_victory_desc_local_player_attacker
			}
			# Desc for player defender
			triggered_desc = { 
				trigger = {
					scope:defender = {
						is_local_player = yes
					}
				}
				desc = kuzhu_install_puppet_war_victory_desc_local_player_defender
			}
			# Desc for a third party involved
			desc = kuzhu_install_puppet_war_victory_desc
		}
	}

	on_victory = {

		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain from winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		
		scope:attacker.joined_faction = {
			every_faction_member = {
				if = {
					limit = {
						government_has_flag = government_has_influence
					}
					change_influence = 1000
				}
				else = {
					add_prestige = 1000
				}	
			}
		}
		scope:attacker = {
			add_prestige = medium_prestige_value
			add_trait = kuzhu_empire_overseer			
		}
		scope:defender = {
			add_prestige = {
				value = medium_prestige_value
				multiply = -1
			}
			#加入被架空的效果
			add_trait = kuzhu_puppet_ruler			
			if = {
				limit = {
					has_realm_law = grand_chancellor_law
				}
				title:e_minister_chancellor = {
					create_title_and_vassal_change = {
						type = conquest
						save_scope_as = change
						add_claim_on_loss = no
					}				
					change_title_holder_include_vassals = {
						holder = scope:attacker
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
				assign_councillor_type = {
					type = councillor_chancellor
		  			target = scope:attacker
					remove_existing_councillor = yes
				}
				hidden_effect = {
					scope:attacker = {
						force_step_down_landed_titles = yes
						block_firing_councillor_effect = { COURT_OWNER = scope:defender }
					}
				}	
			}
			else_if = {
				limit = {
					has_realm_law = grand_marshal_law
				}
				title:e_minister_grand_marshal = {
					create_title_and_vassal_change = {
						type = conquest
						save_scope_as = change
						add_claim_on_loss = no
					}				
					change_title_holder_include_vassals = {
						holder = scope:attacker
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
				assign_councillor_type = {
					type = minister_grand_marshal
		  			target = scope:attacker
					remove_existing_councillor = yes
				}
				hidden_effect = {
					scope:attacker = {
						force_step_down_landed_titles = yes
						block_firing_councillor_effect = { COURT_OWNER = scope:defender }
					}
				}	
			}
			else_if = {
				limit = {
					has_realm_law = grand_censor_law
				}
				title:e_minister_censor = {
					create_title_and_vassal_change = {
						type = conquest
						save_scope_as = change
						add_claim_on_loss = no
					}				
					change_title_holder_include_vassals = {
						holder = scope:attacker
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
				assign_councillor_type = {
					type = councillor_spymaster
		  			target = scope:attacker
					remove_existing_councillor = yes
				}
				hidden_effect = {
					scope:attacker = {
						force_step_down_landed_titles = yes
						block_firing_councillor_effect = { COURT_OWNER = scope:defender }
					}
				}				
			}			
			else = {
				set_diarch = scope:attacker
			}
			try_start_diarchy = regency
			set_diarchy_swing = 100
		}

		# LEGITIMACY FROM LOSING FACTION WAR
		faction_war_end_defeat_legitimacy_effect = yes

		#Mandalas gain or lose piety/devotion depending on Decree
		mandala_war_victory_effects = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = kuzhu_install_puppet_faction_war_white_peace_defender_desc
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
					}
				}
				desc = kuzhu_install_puppet_faction_war_white_peace_attacker_desc
			}
			desc = kuzhu_install_puppet_faction_war_white_peace_desc
		}
		desc = kuzhu_install_puppet_faction_war_white_peace_end_desc
	}

	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			stress_impact = {
 				ambitious = medium_stress_impact_gain
 				arrogant = medium_stress_impact_gain
 			}
		}
		hidden_effect = {
			scope:attacker = {
				add_truce_both_ways = {
					character = scope:defender
					days = 1825
					war = root.war
					result = white_peace
				}				
			}
		}

		scope:defender = {
			stress_impact = {
 				arrogant = medium_stress_impact_gain
 			}			
			add_character_flag = {
				flag = kuzhu_recent_install_puppet_faction_war
				years = faction_nation_fracturing_war_white_peace_cooldown
			}
			add_prestige = minor_prestige_value
		}

		on_white_peace_faction_revolt_war = yes
	}

	on_defeat_desc = {

			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = kuzhu_install_puppet_war_defeat_defender_desc
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
					}
				}
				desc = kuzhu_install_puppet_war_defeat_attacker_desc
			}
			desc = kuzhu_install_puppet_war_defeat_desc
		}
		desc = kuzhu_install_puppet_war_defeat_end_desc
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			add_character_flag = {
				flag = kuzhu_recent_install_puppet_faction_war
				years = faction_nation_fracturing_war_defeat_cooldown
			}
			add_dread = medium_dread_gain
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
