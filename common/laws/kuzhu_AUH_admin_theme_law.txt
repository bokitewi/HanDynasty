#给行政制做军区法

kuzhu_admin_recruit_laws = {
	default = kuzhu_admin_recruit_military_noble
	cumulative = no
	flag = realm_law

	#苦竹：军事贵族制法律
	kuzhu_admin_recruit_military_noble = {
		can_keep = {
			is_independent_ruler = yes
			highest_held_title_tier >= tier_kingdom
			government_allows = administrative
		}

		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = 20
			}
			add_character_modifier = {
				modifier = kuzhu_realm_disorder_modifier
				years = 5
			}				
		}
		can_pass = {
			custom_tooltip = {
				text = kuzhu_need_enough_authority
				NOT = {
					has_realm_law = imperial_bureaucracy_0
				}
			}
			custom_tooltip = {
				text = admin_law_cooldown_desc
				NOT = { has_variable = admin_law_cooldown }
			}			
		}
		pass_cost = {
			prestige = {
				add = increase_crown_authority_prestige_cost
			}
		}	
		modifier = {
			men_at_arms_title_cap = -3
			men_at_arms_title_limit = -3
			men_at_arms_cap = 2
			men_at_arms_limit = 2
			men_at_arms_maintenance = -0.9
			men_at_arms_recruitment_cost = -0.6
			#消除几乎所有头衔兵，加2私兵
		}
		#苦竹：根据年代和情况决定，经济君主喜爱这个
		ai_will_do = {
			value = 0
			if = {
				limit = {
					current_year >= 1150
				}
				add = 3
			}
			if = {
				limit = {
					monthly_character_treasury_income < 0
				}
				add = 3
			}
			if = {
				limit = {
					ai_has_builder_or_pious_builder_personality = yes
				}
				add = 3
			}			
		}
	}
	
	#苦竹：两制并行法律
	kuzhu_admin_recruit_balanced = {
		can_keep = {
			is_independent_ruler = yes
			highest_held_title_tier >= tier_kingdom
			government_allows = administrative
			OR = {
				monthly_character_treasury_income > 0
				treasury > 0 
			}
		}
		can_pass = {
			custom_tooltip = {
				text = kuzhu_need_enough_authority
				NOT = {
					has_realm_law = imperial_bureaucracy_0
				}
			}
			custom_tooltip = {
				text = admin_law_cooldown_desc
				NOT = { has_variable = admin_law_cooldown }
			}			
		}		
		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = 20
			}
			add_character_modifier = {
				modifier = kuzhu_realm_disorder_modifier
				years = 5
			}				
		}
		pass_cost = {
			prestige = {
				add = increase_crown_authority_prestige_cost
			}
		}	
		modifier = {
			men_at_arms_title_cap = -2
			men_at_arms_title_limit = -2
			men_at_arms_cap = 1
			men_at_arms_limit = 1			
			men_at_arms_maintenance = -0.6
			men_at_arms_recruitment_cost = -0.4
			#封臣私兵加1，头衔兵减2
		}
		#苦竹：根据年代和情况决定
		ai_will_do = {
			value = 0
			if = {
				limit = {
					current_year >= 1000
					current_year < 1150
				}
				add = 3			
			}
			if = {
				limit = {
					monthly_character_treasury_income > 0
				}
				add = 3
			}				
		}
	}

	#苦竹：军区制法律
	kuzhu_admin_recruit_theme = {

					has_realm_law = celestial_bureaucracy_3
					has_realm_law = meritocratic_bureaucracy_2

				}
			}
			custom_tooltip = {
				text = admin_law_cooldown_desc

			}			
		}
		pass_cost = {
			prestige = {
				add = increase_crown_authority_prestige_cost
			}
		}	
		modifier = {
			men_at_arms_cap = -1
			men_at_arms_limit = -1
			men_at_arms_maintenance = -0.3
			men_at_arms_recruitment_cost = -0.2
			#封臣头衔兵增加1，私兵减少2
		}
		
		ai_will_do = {
			value = 0
			if = {
				limit = {
					current_year < 1000
				}
				add = 3				
			}
			if = {
				limit = {
					monthly_character_treasury_income > 100
				}
				add = 3
			}
			if = {
				limit = {
					ai_has_warlike_personality = yes
				}
				add = 3
			}				
		}
	}

					has_realm_law = meritocratic_bureaucracy_2
					has_realm_law = meritocratic_bureaucracy_3
				}
			}
			custom_tooltip = {
				text = admin_law_cooldown_desc
				NOT = { has_variable = admin_law_cooldown }
			}
			custom_tooltip = {
				text = kuzhu_need_china_hegemon
				has_title = title:h_china
			}
		}
		pass_cost = {
			prestige = {
				add = increase_crown_authority_prestige_cost
			}
		}	
		modifier = {
		}
		
		ai_will_do = {
			value = 0
			if = {
				limit = {
					monthly_character_treasury_income > 200
				}
				add = 3
			}
			if = {
				limit = {
					ai_has_warlike_personality = yes
				}
				add = 3
			}
			if = {
				limit = {
					has_title = title:h_china
				}
				add = 3
			}					
		}
	}

kuzhu_admin_recruit_vassal_laws = {
	default = kuzhu_admin_recruit_vassal_military_noble
	cumulative = no
	flag = realm_law

	#苦竹：普罗尼亚制法律
	kuzhu_admin_recruit_vassal_military_noble = {
		can_keep = {
			government_allows = administrative
			is_independent_ruler = no
			highest_held_title_tier >= tier_duchy
			trigger_if = {
				limit = {
					liege = { is_independent_ruler = yes }
				}
				liege = { has_realm_law = kuzhu_admin_recruit_military_noble }
			}
			trigger_else = {
				liege = { has_realm_law = kuzhu_admin_recruit_vassal_military_noble }
			}
		}
		can_pass = {
			trigger_if = {
				limit = {
					liege = {
						OR = {
							has_realm_law = kuzhu_admin_recruit_vassal_military_noble
							has_realm_law = kuzhu_admin_recruit_military_noble
						}
					}
				}
				custom_tooltip = {
					text = imperial_bureaucracy_pending_law_change
					always = no
				}
			}
		}
		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = 20
			}
		}	
		modifier = {
			men_at_arms_title_cap = -3
			men_at_arms_title_limit = -3
			men_at_arms_cap = 2
			men_at_arms_limit = 2
