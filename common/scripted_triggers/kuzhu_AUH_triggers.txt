kuzhu_can_be_warlord_trigger = {
	is_landed = yes	
	highest_held_title_tier >= tier_duchy	
	liege = top_liege	
	#忠诚相关
	OR = { 
		has_trait = ambitious
		has_trait = disloyal
		has_trait = arrogant
		has_trait = arbitrary
		has_trait = greedy
		has_trait = deceitful
	}					
	#对ai额外限制
	trigger_if = { 
		limit = { is_ai = yes }
		#与君主好感度够高则不会成为军阀
		opinion = {
			target = top_liege
			value < 80
		}			
		# Prevent ai flipflopping back after demand admin gov interaction
		NOR = {
			var:demand_admin_accepted_ai_block ?= liege
			var:demand_admin_accepted_ai_block ?= top_liege
		}
		
		#要么族长是军阀
		trigger_if = { 
			limit = {
				house.house_head ?= {
					government_has_flag = government_is_kuzhu_warlord
				}
			}
			always = yes
		}
		#如果族长不是军阀，以下全都满足则不可创建：
		#族长是真人
		#族长是世族
		#是族长的亲属
		#是族长的继承人
		trigger_else = { 
			NOT = {
				house.house_head ?= {
					is_ai = no
					any_held_title = { is_noble_family_title = yes }
					is_close_family_of = root
					player_heir = root
				}
			}
		}
	}		
}
