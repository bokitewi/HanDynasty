	is_ai = no

}

rp_has_rewardable_merit_trigger = {
	has_variable = rp_merit_score
	var:rp_merit_score >= 30
}

rp_can_request_merit_reward_trigger = {
	rp_has_rewardable_merit_trigger = yes
	OR = {
		AND = {
			highest_held_title_tier >= tier_kingdom
			var:rp_merit_score >= 120
		}
		AND = {
			highest_held_title_tier = tier_duchy
			var:rp_merit_score >= 90
		}
		AND = {
			highest_held_title_tier = tier_county
			var:rp_merit_score >= 60
		}
		AND = {
			highest_held_title_tier < tier_county
			has_variable = rp_founding_seniority
			var:rp_founding_seniority >= 40
			var:rp_merit_score >= 90
		}
		AND = {
			highest_held_title_tier < tier_county
			has_variable = rp_founding_seniority
			var:rp_founding_seniority >= 20
			var:rp_merit_score >= 60
		}
		AND = {
			highest_held_title_tier < tier_county
			NOR = {
				is_powerful_vassal = yes
				has_trait = ambitious
				has_trait = greedy
				has_trait = arrogant
			}
			OR = {
				NOT = { has_variable = rp_founding_seniority }
				var:rp_founding_seniority < 30
			}
			var:rp_merit_score >= 30
		}
		AND = {
			highest_held_title_tier < tier_county
			OR = {
				is_powerful_vassal = yes
				has_trait = ambitious
				has_trait = greedy
				has_trait = arrogant
			}
			var:rp_merit_score >= 60
		}
	}
}

		highest_held_title_tier >= tier_duchy
		AND = {
			has_variable = rp_merit_score
			var:rp_merit_score >= 100
		}
		AND = {
			has_character_flag = rp_founding_follower
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait

		}
	}

	any_held_title = { tier = tier_county }
	opinion = { target = $LIEGE$ value <= -60 }
	opinion = { target = $TARGET$ value >= 50 }
	current_military_strength >= {
		value = $LIEGE$.current_military_strength
		multiply = 0.08
	}
	$WAR$ = {
		OR = {
			AND = {
				primary_attacker = $LIEGE$
				primary_defender = $TARGET$
			}
			AND = {
				primary_defender = $LIEGE$
				primary_attacker = $TARGET$
			}
		}
		OR = {
			NOT = { has_variable = rp_war_defection_count }
			var:rp_war_defection_count < 2
		}
	}
}

	}
