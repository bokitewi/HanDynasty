### Han Dynasty: repaired settled defensive leagues ###

# A normal, global defensive league.  The workshop implementation referenced
# this type but never supplied it; keeping it in the project makes the feature
# self-contained and leaves nomadic_confederation untouched.
hd_defensive_league_type = {
	on_member_character_joined = {
		save_scope_as = hd_joined_league
		scope:character = {
			if = {
				limit = { NOT = { has_character_modifier = hd_league_member_modifier } }
				add_character_modifier = { modifier = hd_league_member_modifier years = 5 }
			}
		}
		if = {
			limit = { NOT = { has_variable = hd_league_creation_year } }
			set_variable = { name = hd_league_creation_year value = current_year }
			set_variable = { name = hd_league_next_election_year value = current_year }
			change_variable = { name = hd_league_next_election_year add = 5 }
			set_variable = { name = hd_league_unification_year value = current_year }
			change_variable = { name = hd_league_unification_year add = 15 }
		}
		every_player = {
			limit = { confederation = scope:hd_joined_league }
			send_interface_message = {
				type = msg_joined_confederation
				title = msg_joined_confederation
				desc = msg_joined_confederation_desc
				left_icon = scope:character
			}
		}
	}

	on_member_character_left = {
		save_scope_as = hd_left_league
		scope:character = {
		if = {
			limit = { has_character_modifier = hd_league_member_modifier }
			remove_character_modifier = hd_league_member_modifier
		}
		remove_variable = hd_league_vote
		remove_variable = hd_league_forced_vote
		remove_variable = hd_league_leader
	}
		every_player = {
			limit = {
				confederation = scope:hd_left_league
				this != scope:character
			}
			send_interface_message = {
				type = msg_broken_confederation
				title = msg_broken_confederation_no_migrate_root
				desc = msg_broken_confederation_desc_no_migrate_root
				left_icon = scope:character
			}
		}
	}
}

# The anti-regent league uses the same engine hooks but is permanently
# controlled by the event chain rather than by ordinary elections.
hd_anti_regent_league_type = {
	on_member_character_joined = {
		save_scope_as = hd_joined_league
		scope:character = {
			add_character_flag = hd_anti_regent_league_member
		}
		every_player = {
			limit = { confederation = scope:hd_joined_league }
			send_interface_message = {
				type = msg_joined_confederation
				title = msg_joined_confederation
				desc = msg_joined_confederation_desc
				left_icon = scope:character
			}
		}
	}

	on_member_character_left = {
		save_scope_as = hd_left_league
		scope:character = {
			remove_character_flag = hd_anti_regent_league_member
			remove_variable = hd_anti_regent_league_vote
			remove_variable = hd_anti_regent_league_leader
		}
	}
}
