on_combat_end_winner = {
	on_actions = {
		rp_on_combat_end_winner
		rp_on_combat_end_winner_player_subject
		rp_on_combat_end_winner_campaign_incidents
		rp_on_combat_end_winner_ai_campaign_incidents
	}
}

rp_on_combat_end_winner = {
	trigger = {
		exists = combat.combat_war
		OR = {
			combat.combat_war = {
				primary_attacker = { rp_is_merit_activity_enabled_trigger = yes }
			}
			combat.combat_war = {
				primary_defender = { rp_is_merit_activity_enabled_trigger = yes }
			}
		}
	}
	events = {
		rp_merit.0001
	}
}

rp_on_combat_end_winner_player_subject = {
	trigger = {
		exists = combat.combat_war
		side_primary_participant = {
			is_ai = yes
			is_ruler = yes
		}
		OR = {
			side_commander = { is_ai = no }
			any_side_knight = { is_ai = no }
		}
	}
	events = {
		rp_merit.1300
	}
}

rp_on_combat_end_winner_campaign_incidents = {

		OR = {

			}
			combat.combat_war = {
				primary_defender = { rp_is_merit_system_enabled_trigger = yes }
			}
		}
	}

		rp_merit.2000
	}
}

rp_on_combat_end_winner_ai_campaign_incidents = {
	trigger = {
		exists = combat.combat_war
		side_primary_participant = {
			rp_is_merit_activity_enabled_trigger = yes
			is_ai = yes
			NOT = { has_character_flag = rp_campaign_incident_cooldown }
		}
	}
	effect = {
		side_primary_participant = {
			random = {
				chance = 3
				rp_mark_campaign_incident_cooldown_effect = { DAYS = 240 }
				trigger_event = rp_merit_ai.0005
			}
		}
	}
}

rp_on_siege_completion = {
	trigger = {
		exists = root
		exists = scope:war
		root = { is_ruler = yes }
		OR = {
			scope:war = {
				primary_attacker = {
					rp_is_merit_activity_enabled_trigger = yes
					OR = {
						this = root
						this = root.top_liege
					}
				}
			}
			scope:war = {
				primary_defender = {
					rp_is_merit_activity_enabled_trigger = yes
					OR = {
						this = root
						this = root.top_liege
					}

			}

	}
}

					days = 3
				}
			}
		}
	}

rp_on_title_gain_founding_followers = {
	trigger = {
		root = {
			is_ruler = yes
		}
	}
	effect = {
		if = {
			limit = {
				root = {
					rp_is_merit_activity_enabled_trigger = yes
					highest_held_title_tier <= tier_duchy
					NOT = { has_character_flag = rp_recorded_founding_followers }
				}
				scope:title.tier >= tier_duchy
			}
			root = { rp_record_founding_followers_effect = yes }
		}
		root = {
			if = {
				limit = {
					has_character_flag = rp_founding_follower
					scope:title.tier >= tier_county
				}
				rp_establish_founding_house_effect = yes
			}
		}
	}
}

	trigger = {
		root = {
			OR = {
				has_variable = rp_merit_score

				has_character_modifier = rp_danshu_tiequan_modifier

	}

		root = {
