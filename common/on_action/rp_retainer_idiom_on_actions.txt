	on_actions = {
		rp_idiom_on_war_started
		rp_idiom_on_war_started_ai
	}
}

rp_idiom_on_war_started = {
	effect = {
		scope:attacker = {
			if = {
				limit = {
					rp_is_merit_activity_enabled_trigger = yes
					is_ai = no
					NOT = { has_character_flag = rp_idiom_war_started_recently_checked }
					NOT = { has_character_flag = rp_common_flavor_event_cooldown }
					OR = {
						has_relation_friend = scope:defender
						has_relation_best_friend = scope:defender
					}
				}
				scope:defender = { save_scope_as = rp_idiom_enemy }
				rp_mark_common_flavor_event_cooldown_effect = yes
				add_character_flag = { flag = rp_idiom_war_started_recently_checked days = 365 }
				random = { chance = 8 trigger_event = rp_idiom.0001 }
			}
		}
		scope:defender = {
			if = {
				limit = {
					rp_is_merit_activity_enabled_trigger = yes
					is_ai = no
					NOT = { has_character_flag = rp_idiom_war_started_recently_checked }
					NOT = { has_character_flag = rp_common_flavor_event_cooldown }
					OR = {
						has_relation_friend = scope:attacker
						has_relation_best_friend = scope:attacker
					}
				}
				scope:attacker = { save_scope_as = rp_idiom_enemy }
				rp_mark_common_flavor_event_cooldown_effect = yes
				add_character_flag = { flag = rp_idiom_war_started_recently_checked days = 365 }
				random = { chance = 8 trigger_event = rp_idiom.0001 }
			}
		}
	}
}

	on_actions = {
		rp_idiom_on_war_won_attacker
	}
}

rp_idiom_on_war_won_attacker = {
	effect = {
		scope:attacker = {
			if = {
				limit = { rp_is_merit_activity_enabled_trigger = yes }
				scope:defender = { save_scope_as = rp_idiom_enemy }
				rp_idiom_check_revenge_success_effect = yes
				if = {
					limit = { has_character_modifier = rp_idiom_retreat_righteous_modifier }
					remove_character_modifier = rp_idiom_retreat_righteous_modifier
					add_character_modifier = { modifier = rp_idiom_tuibi_sanshe_permanent_modifier years = 100 }
				}
			}
		}
		scope:defender = {
			if = {
				limit = {
					rp_is_merit_activity_enabled_trigger = yes
					is_ai = no
					NOT = { has_character_flag = rp_idiom_defeat_recently_checked }
					NOT = { has_character_flag = rp_common_flavor_event_cooldown }
				}
				scope:attacker = { save_scope_as = rp_idiom_enemy }
				rp_mark_common_flavor_event_cooldown_effect = yes
				add_character_flag = { flag = rp_idiom_defeat_recently_checked days = 365 }
				random = { chance = 15 trigger_event = rp_idiom.0050 }
			}
		}
	}
}
