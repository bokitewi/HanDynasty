### Global maintenance for repaired defensive leagues ###

yearly_global_pulse = {
	on_actions = { hd_confederation_yearly_pulse }
}

hd_confederation_yearly_pulse = {
	effect = {
		every_ruler = {
			if = {
				limit = {
					hd_is_hd_defensive_league = yes
					confederation = {
						has_variable = hd_league_next_election_year
						current_year >= var:hd_league_next_election_year
					}
				}
				confederation = {
					set_variable = { name = hd_last_election_year value = current_year }
					set_variable = { name = hd_league_next_election_year value = current_year }
					change_variable = { name = hd_league_next_election_year add = 5 }
				}
				trigger_event = { id = hd_confederation.0400 days = 1 }
			}
		}
	}
}
