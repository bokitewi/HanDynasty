rp_merit_roster_reward_event_effect = {
	scope = character

	is_valid = {
		exists = scope:rp_merit_target
		scope:rp_merit_target = {
			rp_is_valid_merit_subject_trigger = { LIEGE = root }
			has_variable = rp_merit_score
			var:rp_merit_score >= 30
		}
	}

	effect = {
		scope:rp_merit_target = {
			save_scope_as = rp_merit_reward_target
			save_scope_as = rp_major_battle_petitioner
		}
		trigger_event = rp_merit.0250
	}
}

rp_merit_roster_reward_gold_effect = {
	scope = character

	is_valid = {
		gold >= 50
		exists = scope:rp_merit_target
		scope:rp_merit_target = {
			rp_is_valid_merit_subject_trigger = { LIEGE = root }
			has_variable = rp_merit_score
			var:rp_merit_score >= 30
		}
	}

	effect = {
		remove_short_term_gold = 50
		scope:rp_merit_target = {
			add_gold = 50
			add_opinion = {
				target = root
				modifier = rp_merit_reward_gold_opinion
			}
			rp_consume_merit_effect = { VALUE = 10 }
			add_character_flag = {
				flag = rp_defection_recently_rewarded
				days = 30
			}
		}
		rp_rebuild_merit_roster_ui_list_effect = yes
	}
}
