hd_create_cixiong_shuanggujian_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = hd_cixiong_shuanggujian_name
		description = hd_cixiong_shuanggujian_description
		type = sword
		template = hd_legendary_weapon_template
		visuals = hd_cixiong_shuanggujian
		wealth = scope:wealth
		quality = scope:quality
		max_durability = 100
		history = { type = created_before_history }
		modifier = hd_cixiong_shuanggujian_weapon_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = hd_cixiong_shuanggujian
		equip_artifact_to_owner_replace = yes
	}
}
