### Effects for Cao Cao's exile branch ###

hd_cao_become_wanderer_effect = {
	if = {
		limit = { is_alive = yes }
		if = { limit = { liege ?= { is_alive = yes } } liege = { save_scope_as = hd_cao_wanderer_liege } }
		else = { title:e_han.holder ?= { save_scope_as = hd_cao_wanderer_liege } }
		every_held_title = {
			limit = { tier >= tier_county is_landless_type_title = no is_noble_family_title = no }
			add_to_list = hd_cao_titles_to_return
		}
		if = {
			limit = { exists = scope:hd_cao_wanderer_liege }
			create_title_and_vassal_change = { type = granted save_scope_as = hd_cao_wanderer_return_change add_claim_on_loss = no }
			every_in_list = {
				list = hd_cao_titles_to_return
				change_title_holder_include_vassals = { holder = scope:hd_cao_wanderer_liege change = scope:hd_cao_wanderer_return_change }
			}
			resolve_title_and_vassal_change = scope:hd_cao_wanderer_return_change
		}
		create_landless_adventurer_title_effect = { REASON = flag:wanderer FLAVOR_CHAR = root }
		add_realm_law_skip_effects = camp_purpose_wanderers
		change_government = landless_adventurer_government
	}
}
