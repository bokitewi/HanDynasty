rp_add_merit_effect = {
#tiger-ignore(block, key=scopes)
	if = {
		limit = {
			exists = $LIEGE$
			$LIEGE$ = { is_ai = yes }
		}
		rp_add_ai_merit_credit_effect = { VALUE = $VALUE$ }
	}
	else_if = {
		limit = {
			NOT = { exists = $LIEGE$ }
			is_ai = yes
		}
		rp_add_ai_merit_credit_effect = { VALUE = $VALUE$ }
	}
	else = {
		save_scope_as = rp_merit_recipient

		if = {
			limit = { exists = $LIEGE$ }
			set_variable = {
				name = rp_merit_owner
				value = $LIEGE$
			}
			$LIEGE$ = {
				if = {
					limit = {
						NOT = {
							is_target_in_variable_list = {
								name = rp_merit_roster
								target = scope:rp_merit_recipient
							}
						}
					}
					add_to_variable_list = {
						name = rp_merit_roster
						target = scope:rp_merit_recipient
					}
				}
			}
		}

		if = {
			limit = { NOT = { has_variable = rp_merit_score } }
			set_variable = {
				name = rp_merit_score
				value = 0
			}
		}
		change_variable = {
			name = rp_merit_score
			add = $VALUE$
		}
		if = {
			limit = { var:rp_merit_score > 200 }
			set_variable = {
				name = rp_merit_score
				value = 200
			}
		}
		if = {
			limit = { var:rp_merit_score < -100 }
			set_variable = {
				name = rp_merit_score
				value = -100
			}
		}
		if = {
			limit = {
				var:rp_merit_score > 0
				OR = {
					NOT = { has_variable = rp_merit_peak_score }
					var:rp_merit_score > var:rp_merit_peak_score
				}
			}
			set_variable = {
				name = rp_merit_peak_score
				value = var:rp_merit_score
			}
		}
		rp_update_founding_follower_status_effect = yes
		rp_update_great_house_status_effect = yes
	}
}

rp_convert_hidden_ai_merit_to_player_record_effect = {
	if = {
		limit = {
			is_ai = yes
			exists = $LIEGE$
			$LIEGE$ = { is_ai = no }
			NOT = { has_variable = rp_merit_score }
			OR = {
				has_variable = rp_ai_merit_service_count
				has_character_flag = rp_ai_merit_qualified
			}
		}
		save_scope_as = rp_merit_recipient
		set_variable = {
			name = rp_merit_owner
			value = $LIEGE$
		}
		$LIEGE$ = {
			if = {
				limit = {
					NOT = {
						is_target_in_variable_list = {
							name = rp_merit_roster
							target = scope:rp_merit_recipient
						}
					}
				}
				add_to_variable_list = {
					name = rp_merit_roster
					target = scope:rp_merit_recipient
				}
			}
		}
		if = {
			limit = {
				has_variable = rp_ai_merit_service_count
				var:rp_ai_merit_service_count > 0
			}
			set_variable = {
				name = rp_merit_score
				value = var:rp_ai_merit_service_count
			}
			change_variable = {
				name = rp_merit_score
				multiply = 6
			}
		}
		else = {
			set_variable = {
				name = rp_merit_score
				value = 30
			}
		}
		if = {
			limit = { var:rp_merit_score > 80 }
			set_variable = {
				name = rp_merit_score
				value = 80
			}
		}
		set_variable = {
			name = rp_merit_peak_score
			value = var:rp_merit_score
		}
		remove_variable = rp_ai_merit_service_count
		remove_character_flag = rp_ai_merit_qualified
		rp_update_founding_follower_status_effect = yes
		rp_update_great_house_status_effect = yes
	}
}

rp_consume_merit_effect = {
	if = {
		limit = { is_ai = no }
		if = {
			limit = { NOT = { has_variable = rp_merit_score } }
			set_variable = {
				name = rp_merit_score
				value = 0
			}
		}
		change_variable = {
			name = rp_merit_score
			add = -$VALUE$
		}
		if = {
			limit = { var:rp_merit_score < 0 }
			set_variable = {
				name = rp_merit_score
				value = 0
			}
		}
	}
}

		if = {
			limit = { NOT = { has_variable = rp_founding_seniority } }
			set_variable = { name = rp_founding_seniority value = 0 }
		}
		if = {
			limit = { exists = $LIEGE$ }
			set_variable = { name = rp_founding_liege value = $LIEGE$ }
			rp_add_merit_effect = { LIEGE = $LIEGE$ VALUE = $INITIAL_MERIT$ }
		}
	}

rp_update_founding_follower_status_effect = {
	if = {
		limit = {
			is_ai = yes
			is_independent_ruler = yes
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait
				has_trait = rp_pillar_follower_trait
				has_character_modifier = rp_founding_house_modifier
				has_character_modifier = rp_great_house_modifier
			}
		}
		if = {
			limit = { has_trait = rp_founding_follower_trait }
			remove_trait = rp_founding_follower_trait
		}
		if = {
			limit = { has_trait = rp_core_follower_trait }
			remove_trait = rp_core_follower_trait
		}
		if = {
			limit = { has_trait = rp_pillar_follower_trait }
			remove_trait = rp_pillar_follower_trait
		}
		remove_character_modifier = rp_founding_house_modifier
		remove_character_modifier = rp_great_house_modifier
	}
	if = {
		limit = {
			has_character_flag = rp_founding_follower
			is_independent_ruler = no
		}
		if = {
			limit = {
				has_trait = rp_core_follower_trait
				OR = {
					AND = {
						has_variable = rp_core_follower_years
						var:rp_core_follower_years >= 20
					}
					AND = {
						has_variable = rp_merit_peak_score
						var:rp_merit_peak_score >= 150
					}
					AND = {
						is_ai = yes
						has_character_flag = rp_ai_merit_qualified
					}
				}
			}
			if = {
				limit = { has_trait = rp_founding_follower_trait }
				remove_trait = rp_founding_follower_trait
			}
			if = {
				limit = { has_trait = rp_core_follower_trait }
				remove_trait = rp_core_follower_trait
			}
			if = {
				limit = { NOT = { has_trait = rp_pillar_follower_trait } }
				add_trait = rp_pillar_follower_trait
			}
		}
		else_if = {
			limit = {
				has_trait = rp_founding_follower_trait
				OR = {
					AND = {
						has_variable = rp_founding_trait_years
						var:rp_founding_trait_years >= 15
					}
					AND = {
						has_variable = rp_merit_peak_score
						var:rp_merit_peak_score >= 80
					}
					AND = {
						is_ai = yes
						has_character_flag = rp_ai_merit_qualified
					}
				}
			}
			if = {
				limit = { has_trait = rp_founding_follower_trait }
				remove_trait = rp_founding_follower_trait
			}
			if = {
				limit = { has_trait = rp_pillar_follower_trait }
				remove_trait = rp_pillar_follower_trait
			}
			if = {
				limit = { NOT = { has_trait = rp_core_follower_trait } }
				add_trait = rp_core_follower_trait
			}
			set_variable = { name = rp_core_follower_years value = 0 }
		}
		else_if = {
			limit = {
				NOR = {
					has_trait = rp_founding_follower_trait
					has_trait = rp_core_follower_trait
					has_trait = rp_pillar_follower_trait
				}
				OR = {
					AND = {
						has_variable = rp_founding_court_years
						var:rp_founding_court_years >= 5
					}
					AND = {
						has_variable = rp_founding_duchy_service_years
						var:rp_founding_duchy_service_years >= 5
					}
				}
			}
			if = {
				limit = { has_trait = rp_core_follower_trait }
				remove_trait = rp_core_follower_trait
			}
			if = {
				limit = { has_trait = rp_pillar_follower_trait }
				remove_trait = rp_pillar_follower_trait
			}
			if = {
				limit = { NOT = { has_trait = rp_founding_follower_trait } }
				add_trait = rp_founding_follower_trait
			}
			set_variable = { name = rp_founding_trait_years value = 0 }
		}
		else = {
			if = {
				limit = { has_trait = rp_pillar_follower_trait }
				if = {
					limit = { has_trait = rp_founding_follower_trait }
					remove_trait = rp_founding_follower_trait
				}
				if = {
					limit = { has_trait = rp_core_follower_trait }
					remove_trait = rp_core_follower_trait
				}
			}
			else_if = {
				limit = { has_trait = rp_core_follower_trait }
				if = {
					limit = { has_trait = rp_founding_follower_trait }
					remove_trait = rp_founding_follower_trait
				}
			}
		}
		if = {
			limit = { highest_held_title_tier >= tier_county }
			rp_establish_founding_house_effect = yes
		}
	}
}

rp_establish_founding_house_effect = {
	if = {
		limit = {
			has_character_flag = rp_founding_follower
			is_independent_ruler = no
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait
				has_trait = rp_pillar_follower_trait
			}
			NOT = { has_character_flag = rp_founding_house_established }
		}
		add_character_flag = rp_founding_house_established
		add_character_modifier = {
			modifier = rp_founding_house_modifier
			years = 100
		}
		dynasty = {
			add_dynasty_modifier = {
				modifier = rp_founding_dynasty_modifier
				years = 100
			}
		}
	}
	if = {
		limit = {
			has_character_flag = rp_founding_follower
			is_independent_ruler = no
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait
				has_trait = rp_pillar_follower_trait
			}
			highest_held_title_tier >= tier_county
			OR = {
				highest_held_title_tier >= tier_duchy
				AND = {
					has_variable = rp_founding_seniority
					var:rp_founding_seniority >= 30
				}
			}
			NOT = { has_character_flag = rp_great_house_mark }
		}
		add_character_flag = rp_great_house_mark
		add_character_modifier = {
			modifier = rp_great_house_modifier
			years = 100
		}
	}
}

rp_support_founding_lineage_effect = {
	if = {
		limit = {
			has_character_flag = rp_founding_follower
			is_adult = yes
			is_alive = yes
			is_independent_ruler = no
			NOT = { has_character_flag = rp_founding_lineage_cooldown }
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait
				has_trait = rp_pillar_follower_trait
			}
		}
		add_character_flag = {
			flag = rp_founding_lineage_cooldown
			years = 5
		}
		if = {
			limit = { is_lowborn = yes }
			create_dynasty = {
				spread_to_descendants = yes
			}
		}
		if = {
			limit = { NOT = { has_character_modifier = rp_founding_lineage_modifier } }
			add_character_modifier = {
				modifier = rp_founding_lineage_modifier
				years = 5
			}
		}
		if = {
			limit = {
				is_married = no
				is_betrothed = no
				can_marry_trigger = yes
				is_concubine = no
			}
			random = {
				chance = 70
				if = {
					limit = { is_female = yes }
					create_character = {
						employer = root
						culture = root.culture
						faith = root.faith
						age = 28
						gender_female_chance = 0
						random_traits = yes
						save_scope_as = rp_founding_spouse
					}
				}
				else = {
					create_character = {
						employer = root
						culture = root.culture
						faith = root.faith
						age = 24
						gender_female_chance = 100
						random_traits = yes
						save_scope_as = rp_founding_spouse
					}
				}
				if = {
					limit = {
						exists = scope:rp_founding_spouse
						scope:rp_founding_spouse = {
							is_alive = yes
							is_adult = yes
							is_married = no
							is_betrothed = no
							can_marry_trigger = yes
							is_concubine = no
							could_marry_character_trigger = { CHARACTER = root }
						}
					}
					scope:rp_founding_spouse = {
						add_character_flag = {
							flag = blocked_from_leaving
							years = 10
						}
					}
					marry = scope:rp_founding_spouse
				}
			}
		}
	}
}

rp_set_war_defection_chance_effect = {
	set_variable = { name = rp_defection_personality_mult value = 1 }
	set_variable = { name = rp_defection_situation_mult value = 1 }
	if = { limit = { has_trait = ambitious } change_variable = { name = rp_defection_personality_mult add = 0.15 } }
	if = { limit = { has_trait = greedy } change_variable = { name = rp_defection_personality_mult add = 0.15 } }
	if = { limit = { has_trait = deceitful } change_variable = { name = rp_defection_personality_mult add = 0.15 } }
	if = { limit = { has_trait = craven } change_variable = { name = rp_defection_personality_mult add = 0.15 } }
	if = { limit = { has_trait = arrogant } change_variable = { name = rp_defection_personality_mult add = 0.15 } }
	if = { limit = { has_trait = loyal } change_variable = { name = rp_defection_personality_mult add = -0.20 } }
	if = { limit = { has_trait = honest } change_variable = { name = rp_defection_personality_mult add = -0.20 } }
	if = { limit = { has_trait = brave } change_variable = { name = rp_defection_personality_mult add = -0.20 } }
	if = { limit = { has_trait = compassionate } change_variable = { name = rp_defection_personality_mult add = -0.20 } }
	if = { limit = { has_trait = humble } change_variable = { name = rp_defection_personality_mult add = -0.20 } }
	if = {
		limit = { var:rp_defection_personality_mult < 0.20 }
		set_variable = { name = rp_defection_personality_mult value = 0.20 }
	}
	if = {
		limit = {
			$LIEGE$.current_military_strength <= {
				value = $TARGET$.current_military_strength
				multiply = 0.70
			}
		}
		change_variable = { name = rp_defection_situation_mult add = 0.25 }
	}
	if = {
		limit = {
			$TARGET$.current_military_strength >= {
				value = $LIEGE$.current_military_strength
				multiply = 1.30
			}
		}
		change_variable = { name = rp_defection_situation_mult add = 0.20 }
	}
	if = {
		limit = { rp_has_occupied_county_for_defection_trigger = yes }
		change_variable = { name = rp_defection_situation_mult add = 0.30 }
	}
	if = {
		limit = { rp_defection_same_culture_or_dynasty_trigger = { TARGET = $TARGET$ } }
		change_variable = { name = rp_defection_situation_mult add = 0.20 }
	}
	if = {
		limit = { rp_defection_marriage_tie_trigger = { TARGET = $TARGET$ } }
		change_variable = { name = rp_defection_situation_mult add = 0.25 }
	}
	if = {
		limit = { rp_defection_recently_rebuffed_trigger = { LIEGE = $LIEGE$ } }
		change_variable = { name = rp_defection_situation_mult add = 0.30 }
	}
	if = {
		limit = { rp_defection_recently_humiliated_trigger = { LIEGE = $LIEGE$ } }
		change_variable = { name = rp_defection_situation_mult add = 0.40 }
	}
	if = {
		limit = { opinion = { target = $LIEGE$ value <= -90 } }
		change_variable = { name = rp_defection_situation_mult add = 1.50 }
	}
	else_if = {
		limit = { opinion = { target = $LIEGE$ value <= -75 } }
		change_variable = { name = rp_defection_situation_mult add = 0.30 }
	}
	if = {
		limit = { opinion = { target = $TARGET$ value >= 80 } }
		change_variable = { name = rp_defection_situation_mult add = 0.30 }
	}
	set_variable = { name = rp_defection_chance value = 5 }
	change_variable = { name = rp_defection_chance multiply = var:rp_defection_personality_mult }
	change_variable = { name = rp_defection_chance multiply = var:rp_defection_situation_mult }
	if = {
		limit = {
			OR = {
				has_trait = rp_founding_follower_trait
				has_trait = rp_core_follower_trait
				has_trait = rp_pillar_follower_trait
			}
		}
		change_variable = { name = rp_defection_chance multiply = 0.50 }
	}
	if = {
		limit = { var:rp_defection_chance > 35 }
		set_variable = { name = rp_defection_chance value = 35 }
	}
}

rp_execute_war_defection_effect = {
	scope:rp_defector = {
		add_character_flag = rp_has_defected
		add_character_flag = {
			flag = rp_defection_cooldown
			days = 7300
		}
		remove_variable = rp_defection_likelihood
		add_character_modifier = {
			modifier = rp_traitor_modifier
			years = 20
		}
		add_opinion = {
			target = scope:rp_defection_old_liege
			modifier = rp_defected_from_me_opinion
		}
		add_opinion = {
			target = scope:rp_defection_new_liege
			modifier = rp_defected_to_us_opinion
		}
		if = {
			limit = {
				is_vassal_of = scope:rp_defection_old_liege
				can_create_faction = {
					type = independence_faction
					target = scope:rp_defection_old_liege
				}
			}
			create_faction = {
				type = independence_faction
				target = scope:rp_defection_old_liege
			}
			add_character_modifier = {
				modifier = rp_power_rebel_pressure_modifier
				years = 5
			}
		}
	}
	scope:rp_defection_war = {
		if = {
			limit = { NOT = { has_variable = rp_war_defection_count } }
			set_variable = { name = rp_war_defection_count value = 0 }
		}
		change_variable = { name = rp_war_defection_count add = 1 }
		if = {
			limit = { primary_attacker = scope:rp_defection_old_liege }
			remove_participant = scope:rp_defector
			add_defender = scope:rp_defector
		}
		else_if = {
			limit = { primary_defender = scope:rp_defection_old_liege }
			remove_participant = scope:rp_defector
			add_attacker = scope:rp_defector
		}
	}
	if = {
		limit = { scope:rp_defection_old_liege = { this = root } }
		add_prestige = -150
		add_character_modifier = {
			modifier = rp_campaign_disorder_modifier
			years = 3
		}
		if = {
			limit = { scope:rp_defector = { has_character_flag = rp_hanxin_ambition_actor } }
			scope:rp_defector = { set_variable = { name = hanxin_ambition value = 100 } }
			trigger_event = {
				id = rp_hanxin.0100
				days = 1
			}
		}
		trigger_event = rp_war_defection.0002
	}
	else_if = {
		limit = { scope:rp_defection_new_liege = { this = root } }
		add_prestige = 150
		trigger_event = rp_war_defection.0001
	}
	random = {
		chance = 5
		scope:rp_defection_old_liege = {
			ordered_vassal = {
				limit = {
					NOT = { this = scope:rp_defector }
					rp_is_war_defection_candidate_trigger = {
						LIEGE = scope:rp_defection_old_liege
						TARGET = scope:rp_defection_new_liege
						WAR = scope:rp_defection_war
					}
				}
				order_by = current_military_strength
				if = {
					limit = { NOT = { has_variable = rp_defection_likelihood } }
					set_variable = { name = rp_defection_likelihood value = 0 }
				}
				change_variable = { name = rp_defection_likelihood add = 20 }
				if = {
					limit = { var:rp_defection_likelihood > 100 }
					set_variable = { name = rp_defection_likelihood value = 100 }
				}
			}
		}
	}
}

rp_try_war_defection_for_side_effect = {
	rp_select_war_defection_candidate_effect = {
		LIEGE = $LIEGE$
		TARGET = $TARGET$
		WAR = $WAR$
	}
	if = {
		limit = {
			exists = scope:rp_defector
			scope:rp_defector = {
				rp_is_war_defection_candidate_trigger = {
					LIEGE = $LIEGE$
					TARGET = $TARGET$
					WAR = $WAR$
				}
			}
		}
		$LIEGE$ = { save_scope_as = rp_defection_old_liege }
		$TARGET$ = { save_scope_as = rp_defection_new_liege }
		$WAR$ = { save_scope_as = rp_defection_war }
		scope:rp_defector = {
			rp_set_war_defection_chance_effect = {
				LIEGE = scope:rp_defection_old_liege
				TARGET = scope:rp_defection_new_liege
			}
			if = {
				limit = { rp_defection_recently_rewarded_trigger = { LIEGE = scope:rp_defection_old_liege } }
				add_character_flag = {
					flag = rp_defection_recently_rewarded
					days = 30
				}
				if = {
					limit = { has_variable = rp_defection_likelihood }
					change_variable = { name = rp_defection_likelihood add = -10 }
					if = {
						limit = { var:rp_defection_likelihood < 0 }
						set_variable = { name = rp_defection_likelihood value = 0 }
					}
				}
			}
			else_if = {
				limit = {
					has_variable = rp_defection_likelihood
					var:rp_defection_likelihood >= 100
				}
				root = { rp_execute_war_defection_effect = yes }
			}
			else = {
				random = {
					chance = var:rp_defection_chance
					root = { rp_execute_war_defection_effect = yes }
				}
				if = {
					limit = { NOT = { has_character_flag = rp_has_defected } }
					if = {
						limit = { NOT = { has_variable = rp_defection_likelihood } }
						set_variable = { name = rp_defection_likelihood value = 0 }
					}
					change_variable = { name = rp_defection_likelihood add = 5 }
					if = {
						limit = { var:rp_defection_likelihood > 100 }
						set_variable = { name = rp_defection_likelihood value = 100 }
					}
				}
			}
			rp_clear_war_defection_chance_vars_effect = yes
		}
	}
}

					value = 1
					if = {
						limit = { has_variable = rp_ai_merit_service_count }
						add = var:rp_ai_merit_service_count
					}
				}
			}
			if = {
				limit = { scope:rp_ai_merit_service_count_value > 10 }
				set_variable = { name = rp_ai_merit_service_count value = 10 }
			}
			else = {
				set_variable = {
					name = rp_ai_merit_service_count
					value = scope:rp_ai_merit_service_count_value
				}
			}
			if = {
				limit = { scope:rp_ai_merit_service_count_value >= 8 }
				add_character_flag = rp_ai_merit_qualified
			}
		}
		else = {
			add_character_modifier = {
				modifier = rp_merit_resentful_modifier
				years = 1
			}
			add_character_flag = {
				flag = rp_resentful_mark
				days = 365
			}
		}
	}
}
